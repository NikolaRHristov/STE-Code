#!/usr/bin/env python3
"""Rails Compliance Checker — Scan all output files against the 8 rails."""

import argparse
import json
import os
import re
import sys
from collections import OrderedDict

# Auto-detect project root from __file__ (matching benchmark scripts convention)
PROJECT_ROOT = os.path.abspath(os.path.join(os.path.dirname(__file__), "..", ".."))
STE_CODE = os.path.join(PROJECT_ROOT, "ste-code")

# Fact-check configuration path — load external facts or fall back to defaults
FACTS_CONFIG_PATH = os.path.join(os.path.dirname(__file__), "check-rails-facts.json")

# All known stage directories from the Rails reference (rails.md lines 12-18)
ALL_STAGES = ["extracted", "refined", "merged", "adapted", "artifacts"]


def load_facts(config_path=None):
    """Load fact-check assertions from a JSON config file.

    Falls back to hardcoded defaults if the config file is missing or invalid.
    Each fact is a dict with keys: pattern, negation, expected, desc.
    Multiple facts can be defined; each becomes an R6-Fact-<name> check.
    """
    defaults = {
        "categories": {
            "pattern": "22 categor",
            "negation": "not 22",
            "expected": "19 categories",
            "desc": "Claims 22 categories (should be 19)",
        },
        "model_name": {
            "pattern": r"deepseek-pro(?!-v4-pro)",
            "negation": None,
            "expected": "deepseek-v4-pro",
            "desc": "Wrong model name (should be deepseek-v4-pro)",
        },
        "rule_count": {
            "pattern": "65 rules",
            "negation": None,
            "expected": "53 rules + 4 GR",
            "desc": "Claims 65 rules (should be 53 writing rules + 4 GR rules)",
        },
    }

    if config_path is None:
        config_path = FACTS_CONFIG_PATH

    if os.path.isfile(config_path):
        try:
            with open(config_path) as f:
                loaded = json.load(f)
            # Merge loaded facts over defaults
            merged = dict(defaults)
            merged.update(loaded)
            return merged
        except (json.JSONDecodeError, IOError):
            pass

    return defaults


def build_rails(stages=None, facts=None):
    """Build the rails check registry.

    Args:
        stages: List of stage directories to scan. Default: ALL_STAGES.
        facts: Fact-check assertions dict. Default: load_facts().

    Returns:
        (OrderedDict of rail_name -> rail_config, list of stage directories)
    """
    if stages is None:
        stages = ALL_STAGES
    if facts is None:
        facts = load_facts()

    # Build a regex that matches any known stage directory prefix in file paths
    stages_pattern = "|".join(re.escape(s) for s in stages)
    stage_dir_re = re.compile(rf"^(?:{stages_pattern})/")

    rails = OrderedDict()

    # === R1: Stage Isolation ===
    # Checks that each file lives inside a known stage directory.
    # Previously a permanent no-op (lambda: True). Now does a real check.
    rails["R1-Stage-Isolation"] = {
        "func": lambda content, fpath="": (
            0 if stage_dir_re.search(fpath) else 1
        ),
        "desc": "File is not inside a known stage directory (extracted/refined/merged/adapted/artifacts)",
        "needs_path": True,
    }

    # === R2: Naming Convention ===
    rails["R2-Naming"] = {
        "pattern": r"^(w|r)\d{3}-p\d+-\d+\.md$",
        "desc": "Files must follow [w|r]NNN-pPPPP-PPPP.md pattern",
    }

    # === R3: Completion Integrity ===
    # Three sub-checks: file size, line count, and truncation detection.
    rails["R3-File-Size"] = {
        "func": lambda content: 1 if len(content) < 3000 else 0,
        "desc": "File too small (< 3KB, likely truncated or incomplete)",
    }
    rails["R3-Line-Count"] = {
        "func": lambda content: 1 if content.count("\n") < 30 else 0,
        "desc": "Too few lines (< 30, likely incomplete)",
    }
    rails["R3-Truncation"] = {
        "func": lambda content: (
            1
            if content.rstrip().endswith(
                (",", " the", " and", " in", " to", " of", " a")
            )
            else 0
        ),
        "desc": "File appears truncated (ends mid-sentence)",
    }

    # === R4: Content Fidelity / Fabrication Detection ===
    rails["R4-Fabrication"] = {
        "func": lambda content: sum(
            1
            for t in [
                "This page describes",
                "The key point",
                "In summary",
                "React",
                "Docker",
                "npm",
                "Kubernetes",
            ]
            if t in content
        ),
        "desc": "Fabrication signals detected",
    }

    # === R5: Formatting Standards ===
    # Sub-checks: headings, blanks, boilerplate, STE format, page header
    rails["R5-Headings"] = {
        "func": lambda content: len(
            # Character class [^\n#\s|`>-] matches line starts that are NOT:
            # blank, heading(#), whitespace(\s), table-pipe(|),
            # code-fence(`), blockquote(>), or list-item(-).
            # Backtick is literal inside the bracket — no escape needed in Python.
            re.findall(r"### [^\n]+\n[^\n#\s|`>-]", content)
        ),
        "desc": "No glued headings (blank line required after ###)",
    }
    rails["R5-Blanks"] = {
        "func": lambda content: 1 if "\n\n\n\n" in content else 0,
        "desc": "No triple+ blank lines",
    }
    rails["R5-Boilerplate"] = {
        "func": lambda content: max(
            0, content.count("ASD-STE100 Simplified Technical English") - 4
        ),
        "desc": "Boilerplate header repeated too many times",
    }
    rails["R5-STE-Format"] = {
        "func": lambda content: (
            1
            if "STE:" in content
            and "**STE:**" not in content
            and "> **STE:**" not in content
            else 0
        ),
        "desc": "STE examples not in proper format",
    }
    rails["R5-Page-Header"] = {
        "func": lambda content: (
            0
            if re.match(r"^# Page \d+ of 434", content.split("\n")[0])
            else 1
        ),
        "desc": "Missing # Page N of M header",
    }

    # === R6: Factual Correctness — parameterized from facts config ===
    # Each fact in the config becomes an R6-Fact-<name> check.
    # Uses a factory function to avoid Python closure capture issues
    # inside the loop (late binding of loop variables).
    def make_fact_check(pattern, negation, expected, desc):
        """Build a fact-check closure with correct variable binding."""
        pat = re.compile(pattern) if pattern else None

        def check(content):
            if pat is None:
                return 0
            found = pat.search(content)
            if not found:
                return 0
            if negation:
                neg_pat = re.compile(negation, re.IGNORECASE)
                if neg_pat.search(content):
                    return 0
            return 1

        return check

    for fact_name, fact in facts.items():
        rail_name = f"R6-Fact-{fact_name}"
        check_fn = make_fact_check(
            fact.get("pattern"),
            fact.get("negation"),
            fact.get("expected"),
            fact.get("desc", f"Fact check: {fact_name}"),
        )
        rails[rail_name] = {
            "func": check_fn,
            "desc": fact.get("desc", f"Fact check: {fact_name}"),
        }

    # === R7: Progress Tracking ===
    # Checks that each file is mentioned in PROGRESS.md.
    # NOTE: This is a heuristic check — it looks for filename references.
    # R7 also requires manual verification of PROGRESS.md accuracy.
    rails["R7-Progress-Check"] = {
        "func": lambda content, fpath="": _check_progress(fpath),
        "desc": "File not tracked in PROGRESS.md",
        "needs_path": True,
    }

    # === R8: Error Recovery Signals ===
    # Detects common error patterns that indicate recovery is needed.
    # R8-Wrong-Stage: file naming does not match its stage directory.
    # R8-Model-Reference: wrong model name in file content.
    rails["R8-Wrong-Stage"] = {
        "func": lambda content, fpath="": _check_wrong_stage(fpath),
        "desc": "File in wrong stage directory (naming/stage mismatch)",
        "needs_path": True,
    }
    rails["R8-Model-Reference"] = {
        "func": lambda content: (
            1
            if "deepseek-pro" in content and "deepseek-v4-pro" not in content
            else 0
        ),
        "desc": "Wrong model reference (should be deepseek-v4-pro)",
    }

    return rails, stages


def _check_progress(fpath):
    """Check if a file appears in PROGRESS.md.

    Returns 1 if the file is not tracked, 0 otherwise.
    Skips the check if PROGRESS.md does not exist.
    """
    if not fpath:
        return 0
    progress_path = os.path.join(
        PROJECT_ROOT, ".agents", "state", "PROGRESS.md"
    )
    if not os.path.isfile(progress_path):
        return 0  # No PROGRESS.md to check against

    fname = os.path.basename(fpath)
    try:
        with open(progress_path) as f:
            progress_content = f.read()
        # Check if the exact filename appears in PROGRESS.md
        if fname in progress_content:
            return 0
        # Check if the stage directory is mentioned (batch tracking)
        stage = os.path.basename(os.path.dirname(fpath))
        if stage in progress_content:
            return 0
        return 1
    except Exception:
        return 0


def _check_wrong_stage(fpath):
    """Check if a file's naming convention matches its stage directory.

    Returns 1 if there is a mismatch, 0 otherwise.
    """
    if not fpath:
        return 0

    fname = os.path.basename(fpath)
    stage = os.path.basename(os.path.dirname(fpath))

    # Extraction stage: files must start with 'w'
    if stage == "extracted" and fname.startswith("r"):
        return 1
    # Refinement stage: files must start with 'r'
    if stage == "refined" and fname.startswith("w"):
        return 1
    # Adaptation stage: files must start with 'a-'
    if stage == "adapted" and not fname.startswith("a-"):
        return 1
    # Artifacts stage: files must end with .txt
    if stage == "artifacts" and not fname.endswith(".txt"):
        return 1

    return 0

def check_all(stages=None, rails_filter=None, root=None, fmt="text"):
    """Run all rail checks across specified stages.

    Args:
        stages: List of stage directories to scan. Default: ALL_STAGES.
        rails_filter: List of rail name prefixes to run. Default: all rails.
        root: Project root path. Default: auto-detected from script location.
        fmt: Output format — 'text' or 'json'.

    Returns:
        Number of issues found (exit code 0 = clean, non-zero = issues).
    """
    global PROJECT_ROOT, STE_CODE

    if root:
        PROJECT_ROOT = root
        STE_CODE = os.path.join(PROJECT_ROOT, "ste-code")

    facts = load_facts()
    rails, stages = build_rails(stages=stages, facts=facts)

    # Filter rails if specific rail names or prefixes were requested
    if rails_filter:
        filtered = OrderedDict()
        for name, rail in rails.items():
            for rf in rails_filter:
                if rf in name:
                    filtered[name] = rail
                    break
        rails = filtered

    issues = []
    stats = {"files": 0, "passed": 0, "issues": 0}

    for stage_dir in stages:
        d = os.path.join(STE_CODE, stage_dir)
        if not os.path.isdir(d):
            continue

        for fname in sorted(os.listdir(d)):
            if not fname.endswith(".md"):
                continue
            fpath = os.path.join(d, fname)
            rel_path = os.path.relpath(fpath, STE_CODE)
            stats["files"] += 1
            file_ok = True

            # R2: Naming — checked against filename, not content
            if "R2-Naming" in rails:
                r2 = rails["R2-Naming"]
                if not re.match(r2["pattern"], fname):
                    issues.append(f"🔴 {rel_path}: R2 — {r2['desc']}")
                    file_ok = False

            # Read file content once for all content-based checks
            try:
                with open(fpath) as f:
                    content = f.read()
            except Exception:
                continue

            # Run all content and path-based checks
            for rail_name, rail in rails.items():
                if rail_name == "R2-Naming":
                    continue  # Already checked above

                try:
                    if rail.get("needs_path"):
                        count = rail["func"](content, rel_path)
                    elif "func" in rail:
                        count = rail["func"](content)
                    else:
                        continue
                except Exception:
                    count = 0

                if count > 0:
                    issues.append(
                        f"🟡 {rel_path}: {rail_name} — {rail['desc']} ({count}x)"
                    )
                    stats["issues"] += 1
                    file_ok = False

            if file_ok:
                stats["passed"] += 1

    if fmt == "json":
        result = {"stats": stats, "issues": issues}
        print(json.dumps(result, indent=2))
        return len(issues)

    _print_text_report(stats, issues)
    return len(issues)


def _print_text_report(stats, issues):
    """Print a human-readable text report to stdout."""
    print(f"Files checked: {stats['files']}")
    print(f"Clean: {stats['passed']}")
    print(f"Issues: {stats['issues']}")
    print()

    if issues:
        print("=== Issues Found ===")
        for issue in issues:
            print(f"  {issue}")
        print(f"\n⚠️  {len(issues)} issues need attention.")
    else:
        print("✅ All files pass rails compliance.")

# ---------------------------------------------------------------------------


def main():
    """Parse arguments and run the compliance check."""
    parser = argparse.ArgumentParser(
        description="Rails Compliance Checker — Scan output files against the 8 rails.",
    )
    parser.add_argument(
        "--stages",
        nargs="+",
        default=None,
        choices=ALL_STAGES,
        help="Stage directories to scan (default: all stages)",
    )
    parser.add_argument(
        "--rails",
        nargs="+",
        default=None,
        help="Rail names or prefixes to run (e.g., R2 R5-Headings R6). Default: all rails.",
    )
    parser.add_argument(
        "--root",
        default=None,
        help="Project root path (default: auto-detected from script location)",
    )
    parser.add_argument(
        "--format",
        choices=["text", "json"],
        default="text",
        help="Output format (default: text)",
    )
    parser.add_argument(
        "--list-rails",
        action="store_true",
        help="List all available rail checks and exit.",
    )

    args = parser.parse_args()

    if args.list_rails:
        rails, stages = build_rails()
        print("Available rail checks:")
        print()
        for name, rail in rails.items():
            print(f"  {name}: {rail.get('desc', '(naming check)')}")
        print()
        print(f"Stage directories: {', '.join(stages)}")
        return 0

    return check_all(
        stages=args.stages,
        rails_filter=args.rails,
        root=args.root,
        fmt=args.format,
    )


if __name__ == "__main__":
    sys.exit(main())
