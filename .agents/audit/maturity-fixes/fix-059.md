# STE-Code Benchmark Runner v1.1.0
# Usage: bash .agents/benchmark/run-benchmark.sh [--category readme] [--test bench-001]
# NOTE: This runner executes tests one at a time. For parallel execution (59 tests concurrently),
# use the Python orchestrator: python3 .agents/benchmark/orchestrator.py
# Agent #4 benchmarking orchestrator — runs test cases through hermes -z with STE-Code system prompt

set -euo pipefail

SCRIPT_DIR="$(cd "$(dirname "$0")" && pwd)"
PROJECT_ROOT="$(cd "$SCRIPT_DIR/../.." && pwd)"
SYSTEM_PROMPT="$PROJECT_ROOT/ste-code/artifacts/ste-code-distilled-system-prompt.txt"
TEST_DIR="$SCRIPT_DIR/test-cases"
RESULTS_DIR="$SCRIPT_DIR/results"
TIMESTAMP=$(date -u +"%Y%m%d-%H%M%S")
RUN_DIR="$RESULTS_DIR/run-$TIMESTAMP"

# Colors
RED='\033[0;31m'; GREEN='\033[0;32m'; YELLOW='\033[1;33m'; NC='\033[0m'

# --- CLI Argument Parsing ---
CATEGORY_FILTER=""
TEST_FILTER=""
while [[ $# -gt 0 ]]; do
    case "$1" in
        --category)
            if [ -z "${2:-}" ]; then
                echo -e "${RED}ERROR: --category requires a value (e.g., readme, comments)${NC}"
                exit 1
            fi
            CATEGORY_FILTER="$2"
            shift 2
            ;;
        --test)
            if [ -z "${2:-}" ]; then
                echo -e "${RED}ERROR: --test requires a value (e.g., bench-001)${NC}"
                exit 1
            fi
            TEST_FILTER="$2"
            shift 2
            ;;
        --help|-h)
            echo "Usage: bash .agents/benchmark/run-benchmark.sh [OPTIONS]"
            echo ""
            echo "Run STE-Code benchmark tests sequentially through hermes."
            echo ""
            echo "OPTIONS:"
            echo "  --category NAME   Run only tests in the specified category"
            echo "                    (e.g., readme, comments, error-messages)"
            echo "  --test ID         Run only the specified test (e.g., bench-001)"
            echo "  --help, -h        Show this help message and exit"
            echo ""
            echo "NOTE: This runner is sequential (one test at a time)."
            echo "For parallel execution (all 59 tests run concurrently), use:"
            echo "  python3 .agents/benchmark/orchestrator.py"
            exit 0
            ;;
        *)
            echo -e "${RED}ERROR: Unknown option: $1${NC}"
            echo "Use --help for usage information."
            exit 1
            ;;
    esac
done

echo "=== STE-Code Benchmark Runner ==="
echo "Model: deepseek-v4-pro"
echo "Timestamp: $TIMESTAMP"
if [ -n "$CATEGORY_FILTER" ]; then
    echo "Category filter: $CATEGORY_FILTER"
fi
if [ -n "$TEST_FILTER" ]; then
    echo "Test filter: $TEST_FILTER"
fi
echo ""

# --- Sequential Runner Warning ---
echo -e "${YELLOW}NOTE: This runner executes tests one at a time (sequential).${NC}"
echo -e "${YELLOW}Wall-clock time is approximately 59x slower than the parallel orchestrator.${NC}"
echo -e "${YELLOW}For faster execution, use: python3 .agents/benchmark/orchestrator.py${NC}"
echo ""

# Verify prerequisites
if [ ! -f "$SYSTEM_PROMPT" ]; then
    echo -e "${RED}ERROR: System prompt not found at $SYSTEM_PROMPT${NC}"
    exit 1
fi
echo -e "${GREEN}System prompt found${NC}"

mkdir -p "$RUN_DIR"

# Load all test cases
TEST_FILES=$(ls "$TEST_DIR"/category-*.json 2>/dev/null)
TOTAL_TESTS=$(cat "$TEST_DIR"/category-*.json | grep -c '"id"' || echo 0)
echo "Test cases loaded: $TOTAL_TESTS"
echo ""

# Initialize aggregate results
PASSED=0; FAILED=0
CAT_TRACKING="$RUN_DIR/.cat_tracking"
> "$CAT_TRACKING"
RESULTS_JSON="$RUN_DIR/aggregate-results.json"

# Write JSON header via Python for valid output
python3 -c "
import json, sys
header = {
    'benchmark_id': 'ste-code-v1.1.0',
    'timestamp': '$(date -u +"%Y-%m-%dT%H:%M:%SZ")',
    'model': 'deepseek-v4-pro',
    'total_tests': $TOTAL_TESTS,
    'results': []
}
with open('$RESULTS_JSON', 'w') as f:
    json.dump(header, f, indent=2)
    f.seek(f.tell() - 3)  # Remove trailing ']}' so we can append results
    f.truncate()
"

TEST_INDEX=0

# Process each category
for CAT_FILE in "$TEST_DIR"/category-*.json; do
    CATEGORY=$(basename "$CAT_FILE" .json | sed 's/category-[0-9]-//')

    # Apply category filter
    if [ -n "$CATEGORY_FILTER" ] && [ "$CATEGORY" != "$CATEGORY_FILTER" ]; then
        continue
    fi

    echo "--- Category: $CATEGORY ---"

    # Extract test IDs
    TEST_IDS=$(python3 -c "import json; [print(t['id']) for t in json.load(open('$CAT_FILE'))]")

    for TEST_ID in $TEST_IDS; do
        # Apply test filter
        if [ -n "$TEST_FILTER" ] && [ "$TEST_ID" != "$TEST_FILTER" ]; then
            continue
        fi

        TEST_INDEX=$((TEST_INDEX + 1))
        echo -n "  [$TEST_INDEX/$TOTAL_TESTS] $TEST_ID... "

        # Extract test case data
        TEST_DATA=$(python3 -c "
import json, sys
tests = json.load(open('$CAT_FILE'))
t = next(t for t in tests if t['id'] == '$TEST_ID')
print(t['input'])
print('---PRINCIPLES---')
print(','.join(t['expected_principles']))
print('---FORBIDDEN---')
print(','.join(t.get('forbidden_keywords', [])))
")

        INPUT=$(echo "$TEST_DATA" | sed -n '1,/^---PRINCIPLES---$/p' | sed '$d')
        EXPECTED_PRINCIPLES=$(echo "$TEST_DATA" | sed -n '/^---PRINCIPLES---$/,/^---FORBIDDEN---$/p' | head -2 | tail -1)
        FORBIDDEN=$(echo "$TEST_DATA" | sed -n '/^---FORBIDDEN---$/,$ p' | tail -1)

        # Build full prompt
        FULL_PROMPT="Check the following text for STE-Code compliance. Apply all 14 principles (P1-P14). Replace unapproved terms using the synonym table. Produce ONLY the corrected text with a compliance summary.

$INPUT"

        END_NS=$(python3 -c "import time; print(int(time.time() * 1e9))")

        if [ "$EXIT_OK" = true ]; then
            OUTPUT=$(cat "$RUN_DIR/${TEST_ID}-stdout.txt")
        else
            OUTPUT="HERMES_ERROR"
            echo "HERMES_ERROR" >> "$RUN_DIR/${TEST_ID}-stderr.txt"
        fi

        # Save output (backward-compatible combined file)

        # Score using word-boundary grep to avoid partial matches (e.g., P1 matching P10)
        PRINCIPLES_FOUND=""; PRINCIPLES_MISSED=""; FORBIDDEN_FOUND=""
        CORRECTNESS=1.0

        for p in $(echo "$EXPECTED_PRINCIPLES" | tr ',' ' '); do
            if echo "$OUTPUT" | grep -qiw "$p"; then
                PRINCIPLES_FOUND="$PRINCIPLES_FOUND,$p"
            else
                PRINCIPLES_MISSED="$PRINCIPLES_MISSED,$p"
                CORRECTNESS=$(python3 -c "print(round($CORRECTNESS - 1.0/len('$EXPECTED_PRINCIPLES'.split(',')), 2))")
            fi
        done
        PRINCIPLES_FOUND="${PRINCIPLES_FOUND#,}"; PRINCIPLES_MISSED="${PRINCIPLES_MISSED#,}"

        for fw in $(echo "$FORBIDDEN" | tr ',' ' '); do
            if [ -n "$fw" ] && echo "$OUTPUT" | grep -qiw "$fw"; then
                FORBIDDEN_FOUND="$FORBIDDEN_FOUND,$fw"
                CORRECTNESS=$(python3 -c "print(round(max(0, $CORRECTNESS - 0.1), 2))")
            fi
        done
        FORBIDDEN_FOUND="${FORBIDDEN_FOUND#,}"

        TEST_PASSED=false
        if python3 -c "exit(0 if $CORRECTNESS >= 0.7 else 1)" 2>/dev/null; then
            TEST_PASSED=true; PASSED=$((PASSED + 1))
            echo -e "${GREEN}PASS${NC} (score: $CORRECTNESS, ${LATENCY_MS}ms)"
        else
            FAILED=$((FAILED + 1))
            echo -e "${RED}FAIL${NC} (score: $CORRECTNESS, ${LATENCY_MS}ms)"
        fi

        # Track per-category and per-test data (structured for safe JSON generation)
        echo "$CATEGORY|$TEST_ID|$TEST_PASSED|$CORRECTNESS|$LATENCY_MS|$PRINCIPLES_FOUND|$PRINCIPLES_MISSED|$FORBIDDEN_FOUND" >> "$CAT_TRACKING"
    done
done

# Build aggregate JSON via Python (safe: handles quotes, backslashes, and special characters)
python3 -c "
import json, collections

results = []
cat_data = collections.defaultdict(lambda: {'passed': 0, 'failed': 0, 'scores': []})

with open('$CAT_TRACKING') as f:
    for line in f:
        line = line.strip()
        if not line:
            continue
        parts = line.split('|')
        if len(parts) < 8:
            continue
        cat, test_id, passed_str, score_str, latency_str, found, missed, forbidden = parts[0], parts[1], parts[2], parts[3], parts[4], parts[5], parts[6], parts[7]
        passed = passed_str == 'true'
        score = float(score_str)
        latency = int(latency_str)

        results.append({
            'test_id': test_id,
            'category': cat,
            'correctness_score': score,
            'latency_ms': latency,
            'passed': passed,
            'principles_found': found,
            'principles_missed': missed,
            'forbidden_found': forbidden
        })

        if passed:
            cat_data[cat]['passed'] += 1
        else:
            cat_data[cat]['failed'] += 1
        cat_data[cat]['scores'].append(score)

# Print category breakdown
print()
for cat in sorted(cat_data.keys()):
    d = cat_data[cat]
    avg = sum(d['scores']) / len(d['scores']) if d['scores'] else 0
    print(f'  Category {cat}: {d[\"passed\"]} passed, {d[\"failed\"]} failed, avg score: {avg:.2f}')

# Write final aggregate JSON
total_passed = sum(1 for r in results if r['passed'])
total_failed = sum(1 for r in results if not r['passed'])
total_tests = len(results)
pass_rate = round(total_passed / total_tests * 100, 1) if total_tests > 0 else 0.0

output = {
    'benchmark_id': 'ste-code-v1.1.0',
    'timestamp': '$(date -u +"%Y-%m-%dT%H:%M:%SZ")',
    'model': 'deepseek-v4-pro',
    'total_tests': total_tests,
    'results': results,
    'totals': {
        'passed': total_passed,
        'failed': total_failed,
        'pass_rate': pass_rate
    }
}

with open('$RESULTS_JSON', 'w') as f:
    json.dump(output, f, indent=2)
"

echo ""
echo "============================================"
echo -e "BENCHMARK COMPLETE"
echo -e "Total: $TOTAL_TESTS | ${GREEN}Passed: $PASSED${NC} | ${RED}Failed: $FAILED${NC}"
echo -e "Pass rate: $(python3 -c "print(round($PASSED/$TOTAL_TESTS*100,1))")%"
echo "Results: $RUN_DIR/"
echo "Aggregate: $RESULTS_JSON"
