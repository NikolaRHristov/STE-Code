/**
 * Read user data from the API for a given user ID.
 *
 * @param {string} userId - The unique identifier of the user to read.
 * @returns {Promise<object>} Resolves with the user data object on success.
 *
 * @throws {TypeError}      When `userId` is not a string or is an empty string.
 * @throws {RangeError}     When `userId` exceeds the 128-character limit.
 * @throws {Error}          When the network request fails (no response from server).
 * @throws {Error}          When the server returns an HTTP 404 (user not found).
 * @throws {Error}          When the server returns an HTTP 5xx status (server error).
 * @throws {SyntaxError}    When the response body is not valid JSON.
 */
async function readUserData(userId) {
  const MAX_ID_LENGTH = 128;
  const API_BASE = "https://api.example.com/v2";

  if (typeof userId !== "string" || userId.length === 0) {
    throw new TypeError("userId must be a non-empty string");
  }
  if (userId.length > MAX_ID_LENGTH) {
    throw new RangeError(
      `userId exceeds maximum length of ${MAX_ID_LENGTH} characters`
    );
  }

  let response;
  try {
    response = await fetch(`${API_BASE}/users/${encodeURIComponent(userId)}`);
  } catch {
    throw new Error("Network request failed: no response from server");
  }

  if (response.status === 404) {
    throw new Error(`User not found: ${userId}`);
  }
  if (response.status >= 500) {
    throw new Error(
      `Server error (HTTP ${response.status}) when reading user ${userId}`
    );
  }
  if (!response.ok) {
    throw new Error(
      `Unexpected HTTP ${response.status} when reading user ${userId}`
    );
  }

  let data;
  try {
    data = await response.json();
  } catch {
    throw new SyntaxError("Response body is not valid JSON");
  }

  return data;
}
