@app.route('/api/v1/process', methods=['POST'])
def process_request():
    """Process incoming JSON payload and return the result.

    Returns:
        JSON response with 'status' and 'data' on success,
        or 'error' and 'message' on failure.
    """
    try:
        data = request.get_json()
        if data is None:
            return jsonify({'error': 'Request body must be valid JSON'}), 400
        result = process_data(data)
        return jsonify({'status': 'ok', 'data': result}), 200
    except ValueError as exc:
        return jsonify({'error': str(exc)}), 400
    except Exception as exc:
        # Log the full traceback server-side; never leak it to the client.
        app.logger.exception('Unhandled error in process_request')
        return jsonify({'error': 'Internal server error'}), 500
